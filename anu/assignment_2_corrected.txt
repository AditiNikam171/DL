# %% [MNIST] Feed-forward neural network
import numpy as np
import tensorflow as tf
from tensorflow.keras.datasets import mnist
from tensorflow.keras.layers import Dense, Flatten, Input
from tensorflow.keras.models import Sequential

(x_train, y_train), (x_test, y_test) = mnist.load_data()

# Normalize the pixel values to be between 0 and 1
x_train, x_test = x_train / 255.0, x_test / 255.0

model = Sequential([
    Input(shape=(28, 28)),
    Flatten(),                           # 28x28 -> 784
    Dense(128, activation='relu'),
    Dense(10, activation='softmax')
])

model.compile(optimizer='adam',
              loss='sparse_categorical_crossentropy',
              metrics=['accuracy'])

model.summary()

print("\nTraining the model...")
model.fit(x_train, y_train, epochs=5)

print("\nEvaluating the model...")
test_loss, test_acc = model.evaluate(x_test, y_test, verbose=2)
print(f'\nTest accuracy: {test_acc*100:.2f}%')

predictions = model.predict(x_test[:5])
predicted_class = np.argmax(predictions[0])
true_class = y_test[0]
print("\nPrediction for the first test image:")
print(f"Predicted class: {predicted_class}")
print(f"True class: {true_class}")


# %% [CIFAR-10] Feed-forward neural network
import numpy as np
import tensorflow as tf
from tensorflow.keras.datasets import cifar10
from tensorflow.keras.layers import Dense, Flatten, Input
from tensorflow.keras.models import Sequential

print("--- Stage A: Loading and preparing the CIFAR-10 dataset ---")
(x_train, y_train), (x_test, y_test) = cifar10.load_data()

# Labels come as shape (N, 1); flatten to (N,) for sparse_categorical_crossentropy
y_train, y_test = y_train.flatten(), y_test.flatten()

# Normalize the pixel values to be between 0 and 1
x_train, x_test = x_train / 255.0, x_test / 255.0

print("\n--- Stage B: Defining the Multilayer Perceptron architecture ---")
model = Sequential([
    Input(shape=(32, 32, 3)),
    Flatten(),                           # 32*32*3 = 3072
    Dense(128, activation="relu"),
    Dense(10, activation="softmax"),
])
model.summary()

print("\nCompiling the model...")
model.compile(optimizer="adam",
              loss="sparse_categorical_crossentropy",
              metrics=["accuracy"])

print("\n--- Stage C: Training the model on CIFAR-10 ---")
model.fit(x_train, y_train, epochs=5)

print("\n--- Stage D: Evaluating the model ---")
test_loss, test_acc = model.evaluate(x_test, y_test, verbose=2)
print(f"\nTest accuracy: {test_acc*100:.2f}%")

print("\nMaking predictions on the test set...")
predictions = model.predict(x_test[:5])
predicted_class = np.argmax(predictions[0])
true_class = y_test[0]

print("\nPrediction for the first test image:")
print(f"Predicted class index: {predicted_class}")
print(f"True class index:      {true_class}")

######Viva

• What are the structural differences between the MNIST and CIFAR-10 datasets?
	• Answer: MNIST contains \(28\times28\) grayscale images (1 channel) representing handwritten digits (0–9). CIFAR-10 contains \(32\times32\) color images (3 RGB channels) representing real-world objects (like airplanes, cats, and cars).
• Why did the input dimension of the Flatten layer change from 784 to 3072?
	• Answer: MNIST has \(28 \times 28 \times 1 = 784\) pixels. CIFAR-10 has three color channels, so flattening it results in \(32 \times 32 \times 3 = 3072\) input values.
• What is the purpose of y_train.flatten() in the CIFAR-10 code, and why wasn't it explicitly needed for MNIST?
	• Answer: The Keras CIFAR-10 dataset loader imports labels as a 2D column vector with the shape (50000, 1). We flatten it to a 1D vector (50000,) so it matches the format expected by the sparse_categorical_crossentropy loss function. MNIST labels are already 1D by default.

🏗️ Network Architecture & Design

• Why do both models have exactly 10 neurons in their final dense layer?
	• Answer: The final layer represents the number of target classes. Both MNIST (digits 0–9) and CIFAR-10 (10 distinct object categories) have exactly 10 output classes.
• What is the difference between the hidden layer and the output layer activations in these codes?
	• Answer: The hidden layer uses ReLU to introduce non-linearity and prevent vanishing gradients. The output layer uses Softmax to convert the raw numerical outputs (logits) into a probability distribution that sums up to 1.0.
• Why does a simple Feedforward Neural Network (Dense) yield high accuracy on MNIST but poor accuracy on CIFAR-10?
	• Answer: MNIST digits are simple, centered shapes where pixel spatial locations don't change much. CIFAR-10 features color variations, textures, and objects positioned anywhere in the frame. Flattening a complex image destroys spatial relationships, meaning a Convolutional Neural Network (CNN) is required for CIFAR-10.

⚙️ Loss Functions & Math Optimization

• Why did we choose sparse_categorical_crossentropy instead of regular categorical_crossentropy?
	• Answer: We use sparse categorical crossentropy because our dataset labels are integers (e.g., 3 or 7). If our labels were one-hot encoded vectors (e.g., [0, 0, 0, 1, 0...]), we would use standard categorical_crossentropy.
• What does np.argmax() do when parsing predictions?
	• Answer: The model outputs an array of 10 probabilities. np.argmax() finds and returns the index of the highest probability value, which corresponds directly to the predicted class label.
• What does model.summary() display, and why is it useful?
	• Answer: It displays the structural layout of the model, including the output shape of every layer, the types of layers used, and the total number of trainable and non-trainable parameters.

🚀 Performance & Training

• What is an epoch? How does it differ from a batch?
	• Answer: An epoch is one complete pass of the entire training dataset through the neural network. A batch is a small subset of the dataset (e.g., 32 samples) processed before updating the internal weights.
• What is the Adam optimizer, and why is it preferred over standard SGD?
	• Answer: Adam (Adaptive Moment Estimation) calculates adaptive learning rates for each parameter. It combines the advantages of RMSProp and Momentum, making it faster to converge and less sensitive to initial learning rate settings than standard Stochastic Gradient Descent (SGD).
